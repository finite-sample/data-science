\begin{frame}[fragile]
  \frametitle{Special characters}
  \begin{large_enum}
  \item[-]<1->? - previous atom or group is optional: colou?r - color, colour
  \item[-]<2-> . matches any character except a newline by default\\
          e.g. beg.n matches begun, begin, began
  \item[-]<3->Kleene Operators - named after Stephen Kleene
      \begin{enumerate}
      \item[-]<4->*  matches 0 or more repetitions of the preceding atom or group\\
            e.g. oo*h matches oh, ooh, oooh, etc.\\
            (abc)* will match abc, abcabc, etc.
      \item[-]<5->+  matches 1 or more repetitions of the preceding atom or group\\
            e.g. o+h matches oh, ooh, oooh, etc.
      \end{enumerate}
  \end{large_enum}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Repetition Ranges}
  \begin{enumerate}
  \item[-]<1-> Specific ranges can also be specified
  \item[-]<2-> \small \begin{verbatim}
{ } to specify range for the immediately preceding regex
\end{verbatim}
  \item[-]<3-> \begin{verbatim}
{n}  means exactly n occurrences
\end{verbatim}
  \item[-]<4-> \begin{verbatim}
{n,} means at least n occurrences
\end{verbatim}
  \item[-]<5-> \begin{verbatim}

\end{verbatim}
  \item[-]<6-> Example: \begin{verbatim}
.{0,} is equivalent to .*
\end{verbatim}
  \end{enumerate}
\end{frame}

\frame{
  \frametitle{Edit Distance}
  \begin{large_enum}
    \item[-]<1->Typically refers to minimum edit distance
    \item[-]<2->Minimum number of editing operations to convert one string to another
      \begin{enumerate}
        \item[-]<3->Insertion
        \item[-]<3->Deletion
        \item[-]<3->Substitution
      \end{enumerate}
    \item[-]<3->e.g. two strings: intention, execution
      \begin{enumerate}
        \item[-]<4->align it with second letter
        \item[-]<5->d (delete), s (substitute), s, i(nsert), s
        \item[-]<6->with unit costs, Levenshtein distance = 5
        \item[-]<7->with substitution cost 2 instead of 1, weighted edit distance = 8
      \end{enumerate}
    \item[-]<8->You can implement this at word level so Microsoft Corp. is 1 away from Microsoft.
  \end{large_enum}
}

\frame{
  \frametitle{Text as Data}
  \begin{large_enum}
    \item[-]<1->Bag of words assumption\\\normalsize
          Lose word order
    \item[-]<2->Remove stop words:\\\normalsize
          If, and, but, who, what, the, they, their, a, or, \ldots\\
          \alert{Be careful: one person's stopword is another's key term.}
    \item[-]<3->(Same) Word: Stemming and Lemmatization\\\normalsize
          Porter stemming: taxing $\to$ tax, taxes $\to$ tax, taxation $\to$ taxat, taxable $\to$ taxabl.
    \item[-]<4->Remove rare words\\\normalsize
          Choose thresholds on training data; validate their effect.\\
    \item[-]<5->Convert to lowercase, drop numbers, punctuation, etc.
  \end{large_enum}
}

\begin{frame}[fragile]
\frametitle{How?}
A small example with an explicit stop-word set.
\small
\begin{verbatim}
from nltk.stem import PorterStemmer
from nltk.tokenize import wordpunct_tokenize

text = "Taxing taxes, taxation and taxable income."
stop_words = {"and", "the", "a"}
words = wordpunct_tokenize(text.lower())
words = [w for w in words
         if w.isalpha() and w not in stop_words]
stemmer = PorterStemmer()
stems = [stemmer.stem(w) for w in words]
print(stems)
\end{verbatim}
\end{frame}
\begin{frame}[fragile]
